-- Tabla Usuario
CREATE TABLE IF NOT EXISTS `mydb`.`Usuario` (

);

-- AUDITORÍA PRODUCTO

  idAuditoria INT NOT NULL AUTO_INCREMENT,
  idProducto INT,
